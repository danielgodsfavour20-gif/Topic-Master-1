
import React, { useState, useRef } from 'react';
import { generateLessonText, generateLessonVisual, generateQuiz, generateImage } from './services/geminiService';
import { AppState, LessonContent, QuizQuestion, ImageContent } from './types';
import { LoadingSpinner } from './components/LoadingSpinner';
import { LessonView } from './components/LessonView';
import { QuizView } from './components/QuizView';
import { ImageView } from './components/ImageView';
import { ChatWidget } from './components/ChatWidget';

type AppMode = 'study' | 'image';

const App: React.FC = () => {
  const [state, setState] = useState<AppState>(AppState.IDLE);
  const [mode, setMode] = useState<AppMode>('study');
  const [topic, setTopic] = useState('');
  const [inputTopic, setInputTopic] = useState('');
  
  const [lesson, setLesson] = useState<LessonContent | null>(null);
  const [quizQuestions, setQuizQuestions] = useState<QuizQuestion[]>([]);
  const [imageContent, setImageContent] = useState<ImageContent | null>(null);
  
  const [errorMsg, setErrorMsg] = useState<string | null>(null);
  const searchInputRef = useRef<HTMLInputElement>(null);

  const handleSubmit = async (e?: React.FormEvent, manualTopic?: string) => {
    e?.preventDefault();
    const targetTopic = manualTopic || inputTopic;
    if (!targetTopic.trim()) return;

    setTopic(targetTopic);
    setErrorMsg(null);

    if (mode === 'study') {
      setState(AppState.LOADING_LESSON);
      try {
        // Step 1: Fetch Text First (Fastest)
        const textData = await generateLessonText(targetTopic);
        
        // Immediate Transition to learning screen
        const initialLesson: LessonContent = {
          ...textData as LessonContent,
          isVisualLoading: true
        };
        setLesson(initialLesson);
        setState(AppState.LEARNING);

        // Step 2: Fetch Visual Asynchronously (Background)
        generateLessonVisual(targetTopic).then(svg => {
          setLesson(prev => prev ? { ...prev, svgDiagram: svg, isVisualLoading: false } : null);
        }).catch((err) => {
          console.error("Visual generation failed:", err);
          setLesson(prev => prev ? { ...prev, isVisualLoading: false } : null);
        });

      } catch (err) {
        console.error("Lesson generation failed:", err);
        setErrorMsg("Failed to generate lesson. Please try a different topic.");
        setState(AppState.ERROR);
      }
    } else {
      setState(AppState.LOADING_IMAGE);
      try {
        const imgData = await generateImage(targetTopic);
        setImageContent(imgData);
        setState(AppState.IMAGE_VIEW);
      } catch (err) {
        console.error("Image generation failed:", err);
        setErrorMsg("Failed to generate visual. The model might be busy.");
        setState(AppState.ERROR);
      }
    }
  };

  const handleStartQuiz = async () => {
    if (!lesson) return;
    setState(AppState.LOADING_QUIZ);
    try {
      const context = lesson.explanation.map(s => `${s.title}: ${s.content}`).join("\n");
      const questions = await generateQuiz(lesson.topic, context);
      setQuizQuestions(questions);
      setState(AppState.QUIZ);
    } catch (err) {
      console.error("Quiz generation failed:", err);
      setErrorMsg("Failed to generate quiz.");
      setState(AppState.ERROR);
    }
  };

  const resetApp = () => {
    setInputTopic('');
    setTopic('');
    setLesson(null);
    setQuizQuestions([]);
    setImageContent(null);
    setState(AppState.IDLE);
    setTimeout(() => searchInputRef.current?.focus(), 100);
  };

  const quickTopics = [
    { name: "Black Holes", icon: "🕳️" },
    { name: "Photosynthesis", icon: "🌱" },
    { name: "Blockchain", icon: "🔗" },
    { name: "The Roman Empire", icon: "🏛️" }
  ];

  return (
    <div className="min-h-screen font-sans text-slate-100 flex flex-col bg-[#0f172a] bg-[radial-gradient(ellipse_at_top,_var(--tw-gradient-stops))] from-slate-900 via-[#0f172a] to-black overflow-x-hidden">
      
      {/* Header */}
      <header className="w-full py-6 px-6 md:px-12 flex justify-between items-center z-20 sticky top-0 backdrop-blur-md bg-slate-900/50 border-b border-white/5">
        <div className="flex items-center gap-4 cursor-pointer group" onClick={resetApp}>
          <div className="relative w-10 h-10 transition-transform group-hover:scale-110">
            <div className="absolute inset-0 bg-blue-500/20 rounded-full blur-lg"></div>
            <svg className="w-full h-full" viewBox="0 0 100 100" fill="none">
              <path d="M10 70 C 10 70, 30 80, 50 70 C 70 80, 90 70, 90 70 V 85 C 90 85, 70 95, 50 85 C 30 95, 10 85, 10 85 Z" fill="#1e293b" stroke="#3b82f6" strokeWidth="2" />
              <path d="M35 35 C 35 15, 65 15, 65 35 C 65 48, 55 52, 55 58 H 45 C 45 52, 35 48, 35 35 Z" fill="#fcd34d" fillOpacity="0.2" stroke="#fcd34d" strokeWidth="2" />
            </svg>
          </div>
          <h1 className="text-xl font-heading font-extrabold text-transparent bg-clip-text bg-gradient-to-r from-blue-400 to-indigo-400">
            TOPIC MASTER
          </h1>
        </div>
        {state !== AppState.IDLE && (
          <button 
            onClick={resetApp} 
            className="px-5 py-2 rounded-full bg-slate-800 border border-slate-700 text-slate-300 hover:text-white hover:bg-slate-700 transition-all text-sm font-medium"
          >
            Exit Session
          </button>
        )}
      </header>

      {/* Main Content */}
      <main className="flex-grow flex flex-col items-center justify-center px-4 w-full relative z-0">
        
        {state === AppState.IDLE && (
          <div className="w-full max-w-4xl flex flex-col items-center text-center py-12 animate-fade-in">
              {/* Mode Toggle */}
              <div className="flex p-1 bg-slate-800/80 backdrop-blur-md rounded-2xl mb-12 relative border border-slate-700/50">
                  <button 
                    onClick={() => setMode('study')} 
                    className={`px-8 py-3 rounded-xl font-bold transition-all relative z-10 text-sm md:text-base ${mode === 'study' ? 'text-white' : 'text-slate-400'}`}
                  >
                    Study Lesson
                  </button>
                  <button 
                    onClick={() => setMode('image')} 
                    className={`px-8 py-3 rounded-xl font-bold transition-all relative z-10 text-sm md:text-base ${mode === 'image' ? 'text-white' : 'text-slate-400'}`}
                  >
                    Visual Lab
                  </button>
                  <div className={`absolute top-1 bottom-1 w-[calc(50%-4px)] bg-blue-600 rounded-xl transition-all duration-300 shadow-lg shadow-blue-600/20 ${mode === 'study' ? 'left-1' : 'left-[calc(50%+4px)]'}`}></div>
              </div>

              <h2 className="text-5xl md:text-8xl font-heading font-extrabold mb-6 text-white tracking-tight leading-none">
                {mode === 'study' ? (
                  <>Master <span className="text-transparent bg-clip-text bg-gradient-to-r from-blue-400 to-blue-600">Anything</span>.</>
                ) : (
                  <>See <span className="text-transparent bg-clip-text bg-gradient-to-r from-purple-400 to-indigo-600">Everything</span>.</>
                )}
              </h2>
              
              <p className="text-slate-400 text-lg md:text-xl max-w-2xl mb-12 leading-relaxed">
                {mode === 'study' 
                  ? 'Input any topic and our AI will generate a simplified 5-step interactive lesson with visual diagrams and a final exam.' 
                  : 'Experience topics visually with high-fidelity generated images and detailed context designed for visual learners.'}
              </p>
              
              <form onSubmit={handleSubmit} className="w-full max-w-2xl flex flex-col items-center gap-8">
                <div className="w-full relative group">
                  <input
                    ref={searchInputRef}
                    type="text"
                    autoFocus
                    value={inputTopic}
                    onChange={(e) => setInputTopic(e.target.value)}
                    placeholder="Search a topic..."
                    className="w-full py-8 px-10 rounded-3xl bg-surface/50 backdrop-blur-xl border-2 border-slate-700 text-white focus:outline-none focus:border-blue-500 shadow-2xl transition-all text-2xl md:text-4xl font-bold text-center placeholder-slate-600 group-hover:border-slate-600"
                  />
                  <div className="absolute inset-0 rounded-3xl bg-blue-500/5 -z-10 blur-2xl opacity-0 group-focus-within:opacity-100 transition-opacity"></div>
                </div>

                <button
                  type="submit"
                  disabled={!inputTopic.trim()}
                  className="w-full md:w-auto px-12 py-5 bg-gradient-to-r from-blue-600 to-indigo-600 text-white font-bold text-xl rounded-2xl shadow-xl shadow-blue-900/40 hover:scale-[1.02] active:scale-95 transition-all disabled:opacity-50 disabled:grayscale disabled:scale-100"
                >
                  Generate {mode === 'study' ? 'Lesson' : 'Visual'}
                </button>
              </form>

              {/* Quick Start Topics */}
              <div className="mt-16 w-full max-w-2xl">
                <p className="text-slate-500 text-sm font-bold uppercase tracking-widest mb-6">Popular Explorations</p>
                <div className="flex flex-wrap justify-center gap-3">
                  {quickTopics.map((item) => (
                    <button
                      key={item.name}
                      onClick={() => {
                        setInputTopic(item.name);
                        handleSubmit(undefined, item.name);
                      }}
                      className="px-4 py-2 rounded-full bg-slate-800/50 border border-slate-700/50 text-slate-300 hover:text-white hover:bg-slate-700 transition-all flex items-center gap-2 text-sm"
                    >
                      <span>{item.icon}</span>
                      <span>{item.name}</span>
                    </button>
                  ))}
                </div>
              </div>
          </div>
        )}

        {state === AppState.LOADING_LESSON && <LoadingSpinner message={`Designing your lesson on ${topic}...`} />}
        {state === AppState.LOADING_QUIZ && <LoadingSpinner message={`Building your final exam...`} />}
        {state === AppState.LOADING_IMAGE && <LoadingSpinner message={`Capturing ${topic} in the Lab...`} />}

        {state === AppState.LEARNING && lesson && (
          <LessonView lesson={lesson} onStartQuiz={handleStartQuiz} />
        )}

        {state === AppState.QUIZ && (
          <QuizView 
            questions={quizQuestions} 
            onComplete={() => {}} 
            onRetake={handleStartQuiz} 
            onNewTopic={resetApp} 
          />
        )}

        {state === AppState.IMAGE_VIEW && imageContent && (
          <ImageView content={imageContent} onClose={resetApp} />
        )}

        {state === AppState.ERROR && (
          <div className="text-center p-12 bg-red-500/10 border border-red-500/20 rounded-[2rem] max-w-xl animate-slide-up">
            <div className="w-16 h-16 bg-red-500/20 rounded-full flex items-center justify-center mx-auto mb-6">
              <svg xmlns="http://www.w3.org/2000/svg" className="h-8 w-8 text-red-500" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z" />
              </svg>
            </div>
            <h3 className="text-2xl font-bold text-white mb-2">Something went wrong</h3>
            <p className="text-slate-400 mb-8">{errorMsg || "We couldn't generate that content right now."}</p>
            <button onClick={resetApp} className="px-8 py-3 bg-slate-800 text-white rounded-xl hover:bg-slate-700 transition">
              Back to Home
            </button>
          </div>
        )}
      </main>

      {/* Floating Chat Assistant */}
      <ChatWidget 
        currentTopic={topic || undefined} 
        requireDetails={state === AppState.LEARNING} 
      />

      {/* Background Decor */}
      <div className="fixed top-0 left-0 w-full h-full -z-50 pointer-events-none opacity-40 overflow-hidden">
        <div className="absolute top-[-10%] right-[-10%] w-[50%] h-[50%] bg-blue-600/10 blur-[150px] rounded-full"></div>
        <div className="absolute bottom-[-10%] left-[-10%] w-[50%] h-[50%] bg-indigo-600/10 blur-[150px] rounded-full"></div>
      </div>
    </div>
  );
};

export default App;
