
import React, { useState, useEffect } from 'react';
import { LessonContent } from '../types';

interface LessonViewProps {
  lesson: LessonContent;
  onStartQuiz: () => void;
}

export const LessonView: React.FC<LessonViewProps> = ({ lesson, onStartQuiz }) => {
  const [currentStep, setCurrentStep] = useState(0);
  const [isSpeaking, setIsSpeaking] = useState(false);
  
  const getCleanSvg = (svg?: string) => {
    if (!svg) return "";
    return svg.replace(/```xml/g, '').replace(/```/g, '').trim();
  };

  const nextStep = () => {
    stopSpeaking();
    if (currentStep < lesson.explanation.length - 1) {
      setCurrentStep(prev => prev + 1);
    }
  };

  const prevStep = () => {
    stopSpeaking();
    if (currentStep > 0) {
      setCurrentStep(prev => prev - 1);
    }
  };

  const speakContent = () => {
    if (isSpeaking) {
      stopSpeaking();
      return;
    }
    const step = lesson.explanation[currentStep];
    const utterance = new SpeechSynthesisUtterance(`${step.title}. ${step.content}`);
    utterance.onend = () => setIsSpeaking(false);
    setIsSpeaking(true);
    window.speechSynthesis.speak(utterance);
  };

  const stopSpeaking = () => {
    window.speechSynthesis.cancel();
    setIsSpeaking(false);
  };

  useEffect(() => {
    return () => stopSpeaking();
  }, []);

  return (
    <div className="w-full max-w-6xl mx-auto animate-slide-up pb-20">
      <div className="mb-6 text-center">
        <h1 className="mt-4 text-4xl md:text-5xl font-heading font-bold text-white mb-2">
          {lesson.title}
        </h1>
      </div>

      <div className="flex flex-col lg:flex-row gap-8 items-start">
        
        {/* Left Column: Visual Diagram */}
        <div className="w-full lg:w-2/3 bg-surface rounded-3xl p-4 md:p-6 shadow-2xl border border-slate-700/50 flex flex-col min-h-[400px]">
          <div className="flex items-center justify-between mb-4">
            <h2 className="text-xl font-heading font-bold text-accent flex items-center">
              <svg xmlns="http://www.w3.org/2000/svg" className="h-6 w-6 mr-2" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 16l4.586-4.586a2 2 0 012.828 0L16 16m-2-2l1.586-1.586a2 2 0 012.828 0L20 14m-6-6h.01M6 20h12a2 2 0 002-2V6a2 2 0 00-2-2H6a2 2 0 00-2 2v12a2 2 0 002 2z" />
              </svg>
              Visual Diagram
            </h2>
            {lesson.isVisualLoading && (
              <span className="flex items-center gap-2 text-xs text-blue-400 animate-pulse bg-blue-500/10 px-2 py-1 rounded">
                <div className="w-2 h-2 rounded-full bg-blue-500 animate-bounce"></div>
                Generating Visual...
              </span>
            )}
          </div>
          
          <div className="flex-grow w-full bg-slate-900 rounded-2xl overflow-hidden p-4 border border-slate-700/50 shadow-inner flex items-center justify-center min-h-[300px]">
            {lesson.isVisualLoading || !lesson.svgDiagram ? (
              <div className="flex flex-col items-center justify-center text-slate-600 gap-4 opacity-40">
                <svg className="w-24 h-24 animate-pulse" viewBox="0 0 100 100">
                  <rect x="10" y="10" width="80" height="80" fill="none" stroke="currentColor" strokeWidth="2" strokeDasharray="5,5" />
                  <circle cx="50" cy="50" r="20" fill="none" stroke="currentColor" strokeWidth="2" strokeDasharray="3,3" />
                  <path d="M10 10 L 90 90 M 90 10 L 10 90" stroke="currentColor" strokeWidth="1" strokeDasharray="2,2" />
                </svg>
                <p className="font-mono text-xs tracking-widest uppercase">Building Schematic...</p>
              </div>
            ) : (
              <div 
                className="w-full h-full flex items-center justify-center [&>svg]:w-full [&>svg]:h-full [&>svg]:max-h-[500px] transition-opacity duration-700 animate-fade-in"
                dangerouslySetInnerHTML={{ __html: getCleanSvg(lesson.svgDiagram) }}
              />
            )}
          </div>
        </div>

        {/* Right Column: Interactive Explanation Steps */}
        <div className="w-full lg:w-1/3 flex flex-col gap-6">
          <div className="bg-surface rounded-3xl p-6 md:p-8 shadow-2xl border border-slate-700/50 flex flex-col h-full relative overflow-hidden">
             
             <div className="absolute top-0 left-0 w-full h-1 bg-slate-800">
               <div 
                 className="h-full bg-secondary transition-all duration-300"
                 style={{ width: `${((currentStep + 1) / lesson.explanation.length) * 100}%` }}
               />
             </div>

             <div className="flex justify-between items-center mb-6 mt-2">
               <span className="text-sm font-bold text-slate-400">Step {currentStep + 1} of {lesson.explanation.length}</span>
               <div className="flex gap-2">
                 <button onClick={prevStep} disabled={currentStep === 0} className="p-2 rounded-lg bg-slate-800 hover:bg-slate-700 disabled:opacity-30 transition">
                   <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5 text-white" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 19l-7-7 7-7" /></svg>
                 </button>
                 <button onClick={nextStep} disabled={currentStep === lesson.explanation.length - 1} className="p-2 rounded-lg bg-slate-800 hover:bg-slate-700 disabled:opacity-30 transition">
                   <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5 text-white" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" /></svg>
                 </button>
               </div>
             </div>

             <div className="flex-grow flex flex-col justify-center animate-fade-in" key={currentStep}>
                <div className="flex items-start justify-between gap-4 mb-4">
                  <h3 className="text-2xl font-heading font-bold text-white leading-tight">
                    {lesson.explanation[currentStep].title}
                  </h3>
                  <button 
                    onClick={speakContent}
                    className={`flex-shrink-0 p-3 rounded-full transition-colors ${isSpeaking ? 'bg-primary text-white animate-pulse' : 'bg-slate-800 text-slate-400 hover:bg-slate-700'}`}
                  >
                    <svg xmlns="http://www.w3.org/2000/svg" className="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15.536 8.464a5 5 0 010 7.072m2.828-9.9a9 9 0 010 12.728M5.586 15H4a1 1 0 01-1-1v-4a1 1 0 011-1h1.586l4.707-4.707C10.923 3.663 12 4.109 12 5v14c0 .891-1.077 1.337-1.707.707L5.586 15z" /></svg>
                  </button>
                </div>
                <p className="text-lg text-slate-300 leading-relaxed">
                  {lesson.explanation[currentStep].content}
                </p>
             </div>

             <div className="mt-8 pt-6 border-t border-slate-700/50">
               {currentStep === lesson.explanation.length - 1 ? (
                 <button
                   onClick={onStartQuiz}
                   className="w-full py-4 bg-gradient-to-r from-primary to-secondary text-white font-bold rounded-xl shadow-lg hover:scale-[1.02] transition flex items-center justify-center"
                 >
                   Take Final Exam
                 </button>
               ) : (
                 <button onClick={nextStep} className="w-full py-4 bg-slate-700 hover:bg-slate-600 text-white font-semibold rounded-xl transition">
                   Continue Learning
                 </button>
               )}
             </div>
          </div>
        </div>

      </div>
    </div>
  );
};
